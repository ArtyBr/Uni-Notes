```sql
SELECT roomtype
FROM accommodation a1
WHERE EXISTS (SELECT *
			 FROM halls
			 WHERE campus = 'Coventry'
			 AND a1.hallname = halls.hallname)
AND a1.price < 50
AND NOT EXISTS (SELECT roomtype
				FROM accommodation a2
				WHERE EXISTS (SELECT *
							 FROM halls
							 WHERE campus <> 'Coventry'
							 AND a2.hallname = halls.hallname))
				AND a2.roomtype = a1.roomtype
```
